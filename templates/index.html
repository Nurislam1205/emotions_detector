# Группировка по тональности
POS = {'happiness','love',
       'fun','enthusiasm',
       'relief','surprise'}
NEG = {'sadness','anger',
       'hate','worry',
       'boredom'}

def group(e):
    if e in POS: return 'Positive'
    if e in NEG: return 'Negative'
    return 'Neutral'

df['group'] = (
    df['sentiment'].apply(group))

grp = df['group'].value_counts()
plt.pie(grp.values,
        labels=grp.index,
        autopct='%1.1f%%')
plt.show()

  df.groupby('sentiment')[
    'word_count'
  ].quantile([0.25,0.5,0.75])
  .unstack()
)

for emotion \
    in perc.index:
  p25 = perc.loc[emotion, 0.25]
  med = perc.loc[emotion, 0.50]
  p75 = perc.loc[emotion, 0.75]
  plt.barh(emotion,
           p75-p25, left=p25)
  plt.plot(med, emotion, '|')

plt.title('P25–P75 per Emotion')
plt.show()

Топ-10 самых частых слов
Python код
from collections import Counter
import re

STOP = {'i', 'the', 'a',
        'to', 'and', 'is'}

words = []
for txt in df['content']:
  words += [
    w.lower() for w
    in re.findall(
      r"[a-zA-Z']+", txt)
    if w not in STOP
    and len(w) > 2
  ]

top = Counter(words).most_common(10)
words, cnts = zip(*top)

sns.barplot(x=list(cnts),
            y=list(words))
plt.show()
